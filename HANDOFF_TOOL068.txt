[FORMULA]

fee = commission + fixedFee
settlement = sale + buyerShipping - fee - sellerShipping - otherCost
netProfit = settlement - productCost
netMargin = netProfit / (sale + buyerShipping) * 100
effectiveFeeRate = fee / sale * 100
requiredSale = (productCost + sellerShipping + otherCost + fixedFee + targetProfit - buyerShipping) / (1 - rate/100)
